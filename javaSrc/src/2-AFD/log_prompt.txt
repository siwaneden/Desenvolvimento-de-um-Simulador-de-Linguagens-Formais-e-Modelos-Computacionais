faça um AFD em java

Claro! Aqui está um exemplo de um AFD (Autômato Finito Determinístico) em Java, que reconhece a linguagem de palavras binárias que terminam em 01.

Código Java – AFD

java

import java.util.Scanner;

public class AFD {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int estado = 0;

        System.out.print("Digite uma palavra binária: ");
        String palavra = sc.nextLine();

        for (int i = 0; i < palavra.length(); i++) {

            char simbolo = palavra.charAt(i);

            switch (estado) {

                case 0:
                    if (simbolo == '0') {
                        estado = 1;
                    } else if (simbolo == '1') {
                        estado = 0;
                    } else {
                        estado = -1;
                    }
                    break;

                case 1:
                    if (simbolo == '0') {
                        estado = 1;
                    } else if (simbolo == '1') {
                        estado = 2;
                    } else {
                        estado = -1;
                    }
                    break;

                case 2:
                    if (simbolo == '0') {
                        estado = 1;
                    } else if (simbolo == '1') {
                        estado = 0;
                    } else {
                        estado = -1;
                    }
                    break;
            }

            if (estado == -1) {
                break;
            }
        }

        if (estado == 2) {
            System.out.println("Palavra ACEITA!");
        } else {
            System.out.println("Palavra REJEITADA!");
        }

        sc.close();
    }
}

package modelo;

import java.util.ArrayList;
import java.util.List;

public abstract class Automato {

    protected List<String> alfabeto;
    protected List<Estado> estados;
    protected List<Transicao> transicoes;

    public Automato() {
        this.alfabeto = new ArrayList<>();
        this.estados = new ArrayList<>();
        this.transicoes = new ArrayList<>();
    }

    // Adicionar simbolo ao alfabeto
    public void adicionarSimbolo(String simbolo) {
        if (!alfabeto.contains(simbolo)) {
            alfabeto.add(simbolo);
        }
    }

    // Adicionar estado
    public void adicionarEstado(Estado estado) {
        estados.add(estado);
    }

    // Adicionar transicao
    public void adicionarTransicao(Transicao transicao) {
        transicoes.add(transicao);
    }

    // Buscar estado pelo nome
    public Estado buscarEstado(String nome) {
        for (Estado estado : estados) {
            if (estado.getNome().equals(nome)) {
                return estado;
            }
        }

        return null;
    }

    // Obter estado inicial
    public Estado getEstadoInicial() {
        for (Estado estado : estados) {
            if (estado.isInicial()) {
                return estado;
            }
        }

        return null;
    }

    // Obter estados finais
    public List<Estado> getEstadosFinais() {
        List<Estado> finais = new ArrayList<>();

        for (Estado estado : estados) {
            if (estado.isFinal()) {
                finais.add(estado);
            }
        }

        return finais;
    }

    // Exibir estrutura do automato
    public void exibirAutomato() {

        System.out.println("\n===== AUTOMATO =====");

        System.out.println("Alfabeto: " + alfabeto);

        System.out.println("\nEstados:");

        for (Estado estado : estados) {
            System.out.println(
                estado.getNome()
                + (estado.isInicial() ? " [Inicial]" : "")
                + (estado.isFinal() ? " [Final]" : "")
            );
        }

        System.out.println("\nTransicoes:");

        for (Transicao transicao : transicoes) {
            System.out.println(transicao);
        }
    }

    // Getters
    public List<String> getAlfabeto() {
        return alfabeto;
    }

    public List<Estado> getEstados() {
        return estados;
    }

    public List<Transicao> getTransicoes() {
        return transicoes;
    }

    // Metodo que cada automato implementara
    public abstract boolean processar(String palavra);
}

Agr faça o AFD.java

Agora vamos implementar o AFD.java, utilizando as classes Estado.java, Transicao.java e Automato.java que criamos anteriormente.

O AFD será responsável por percorrer uma palavra, verificar cada símbolo e determinar se ela pertence à linguagem configurada.

1. AFD.java
Crie o arquivo dentro da pasta afd.


java

package afd;

import modelo.Automato;
import modelo.Estado;
import modelo.Transicao;

public class AFD extends Automato {

    public AFD() {
        super();
    }

    // Verifica se a palavra pertence ao alfabeto
    private boolean validarAlfabeto(String palavra) {

        for (int i = 0; i < palavra.length(); i++) {

            String simbolo = String.valueOf(palavra.charAt(i));

            if (!alfabeto.contains(simbolo)) {
                return false;
            }
        }

        return true;
    }

    // Busca a transicao correspondente
    private Estado buscarTransicao(Estado atual, String simbolo) {

        for (Transicao transicao : transicoes) {

            if (transicao.getOrigem().equals(atual)
                    && transicao.getSimbolo().equals(simbolo)) {

                return transicao.getDestino();
            }
        }

        return null;
    }

    // Processa a palavra no AFD
    @Override
    public boolean processar(String palavra) {

        Estado atual = getEstadoInicial();

        // Verifica se existe estado inicial
        if (atual == null) {
            System.out.println("Erro: estado inicial nao definido!");
            return false;
        }

        // Verifica se existem estados finais
        if (getEstadosFinais().isEmpty()) {
            System.out.println("Erro: nenhum estado final definido!");
            return false;
        }

        // Verifica se a palavra pertence ao alfabeto
        if (!validarAlfabeto(palavra)) {
            return false;
        }

        // Percorre cada simbolo da palavra
        for (int i = 0; i < palavra.length(); i++) {

            String simbolo = String.valueOf(palavra.charAt(i));

            Estado proximo = buscarTransicao(atual, simbolo);

            // Se nao existir transicao, rejeita
            if (proximo == null) {
                return false;
            }

            atual = proximo;
        }

        // Verifica se terminou em um estado final
        return atual.isFinal();
    }
}

Agora vamos implementar o AFN.java (Autômato Finito Não Determinístico).

Diferentemente do AFD, o AFN permite:

Mais de uma transição para o mesmo símbolo no mesmo estado.

Transições vazias (ε), que não consomem nenhum símbolo da palavra.

Para isso, vamos utilizar o conceito de conjunto de estados atuais.

Crie o arquivo dentro da pasta afn.

package afn;

import modelo.Automato;

import java.util.ArrayList;

import java.util.List;
import java.util.Set;

public class AFN extends Automato {

    private static final String EPSILON = "ε";

    public AFN() {
        super();
    }

    // Busca todos os destinos de uma transicao
    private Set<Estado> buscarDestinos(
            Estado atual, String simbolo) {

        Set<Estado> destinos = new HashSet<>();

        for (Transicao transicao : transicoes) {

            if (transicao.getOrigem().equals(atual)
                    && transicao.getSimbolo().equals(simbolo)) {

                destinos.add(transicao.getDestino());
            }
        }

        return destinos;
    }

    // Calcula o fechamento epsilon
    private Set<Estado> fechamentoEpsilon(Set<Estado> estadosAtuais) {

        Set<Estado> fechamento = new HashSet<>(estadosAtuais);

        List<Estado> fila = new ArrayList<>(estadosAtuais);

        while (!fila.isEmpty()) {

            Estado atual = fila.remove(0);

            Set<Estado> destinos = buscarDestinos(atual, EPSILON);

            for (Estado destino : destinos) {

                if (fechamento.add(destino)) {
                    fila.add(destino);
                }
            }
        }

        return fechamento;
    }

    // Processa a palavra no AFN
    @Override
    public boolean processar(String palavra) {

        Estado inicial = getEstadoInicial();

        if (inicial == null) {
            System.out.println("Erro: estado inicial nao definido!");
            return false;
        }

        if (getEstadosFinais().isEmpty()) {
            System.out.println("Erro: nenhum estado final definido!");
            return false;
        }

        // Verifica se todos os simbolos pertencem ao alfabeto
        for (int i = 0; i < palavra.length(); i++) {

            String simbolo = String.valueOf(palavra.charAt(i));

            if (!alfabeto.contains(simbolo)) {
                return false;
            }
        }

        // Comeca pelo estado inicial
        Set<Estado> estadosAtuais = new HashSet<>();
        estadosAtuais.add(inicial);

        estadosAtuais = fechamentoEpsilon(estadosAtuais);

        // Percorre a palavra
        for (int i = 0; i < palavra.length(); i++) {

            String simbolo = String.valueOf(palavra.charAt(i));

            Set<Estado> proximosEstados = new HashSet<>();

            // Analisa todos os estados atuais
            for (Estado estado : estadosAtuais) {

                proximosEstados.addAll(
                    buscarDestinos(estado, simbolo)
                );
            }

            // Aplica fechamento epsilon novamente
            estadosAtuais = fechamentoEpsilon(proximosEstados);

            // Se nao houver caminhos, rejeita
            if (estadosAtuais.isEmpty()) {
                return false;
            }
        }

        // Verifica se algum caminho terminou em estado final
        estadosAtuais = fechamentoEpsilon(estadosAtuais);

        for (Estado estado : estadosAtuais) {

            if (estado.isFinal()) {
                return true;
            }
        }

        return false;
    }
}
2. Entendendo a diferença entre AFD e AFN
Imagine que temos o estado q0 e o símbolo a.

No AFD:

Uma única transição

q0
q1
Ao ler a, q0 obrigatoriamente vai para q1.

No AFN:

Múltiplas transições

q0
q1
q2
Ao ler a, o AFN pode seguir pelos dois caminhos.

O nosso código utiliza HashSet para armazenar os estados possíveis, evitando duplicações.

    public TransicaoPilha(
            Estado origem,

            String topoPilha,
            String substituirPilha,
            Estado destino) {

        this.origem = origem;
        this.simboloEntrada = simboloEntrada;
        this.topoPilha = topoPilha;
        this.substituirPilha = substituirPilha;
        this.destino = destino;
    }

    public Estado getOrigem() {
        return origem;
    }

    public String getSimboloEntrada() {
        return simboloEntrada;
    }

    public String getTopoPilha() {
        return topoPilha;
    }

    public String getSubstituirPilha() {
        return substituirPilha;
    }

    public Estado getDestino() {
        return destino;
    }

    @Override
    public String toString() {
        return origem.getNome()
                + " -- (" + simboloEntrada
                + ", " + topoPilha
                + " / " + substituirPilha
                + ") --> " + destino.getNome();
    }

4. AutomatoPilha.java
Agora vamos criar a classe principal do AP.

Ela herdará de Automato.java, mas terá uma lista própria de transições, porque as transições do AP possuem informações adicionais.


java

package ap;

import modelo.Automato;
import modelo.Estado;

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;

public class AutomatoPilha extends Automato {

    private static final String EPSILON = "ε";

    private List<TransicaoPilha> transicoesPilha;
    private String simboloInicialPilha;

    public AutomatoPilha() {
        super();
        transicoesPilha = new ArrayList<>();
    }

    // Define o simbolo inicial da pilha
    public void setSimboloInicialPilha(String simbolo) {
        this.simboloInicialPilha = simbolo;
    }

    // Adiciona uma transicao
    public void adicionarTransicaoPilha(
            TransicaoPilha transicao) {

        transicoesPilha.add(transicao);
    }

    // Exibe as transicoes do AP
    public void exibirTransicoesPilha() {

        System.out.println("\n===== TRANSICOES DO AP =====");

        for (TransicaoPilha transicao : transicoesPilha) {
            System.out.println(transicao);
        }
    }

    // Verifica se a palavra pertence ao alfabeto
    private boolean validarPalavra(String palavra) {

        for (int i = 0; i < palavra.length(); i++) {

            String simbolo =
                    String.valueOf(palavra.charAt(i));

            if (!alfabeto.contains(simbolo)) {
                return false;
            }
        }

        return true;
    }

    // Processa a palavra
    @Override
    public boolean processar(String palavra) {

        Estado atual = getEstadoInicial();

        if (atual == null) {
            System.out.println("Estado inicial nao definido!");
            return false;
        }

        if (getEstadosFinais().isEmpty()) {
            System.out.println("Nenhum estado final definido!");
            return false;
        }

        if (simboloInicialPilha == null
                || simboloInicialPilha.isEmpty()) {

            System.out.println("Simbolo inicial da pilha nao definido!");
            return false;
        }

        if (!validarPalavra(palavra)) {
            return false;
        }

        // Cria a pilha
        Deque<String> pilha = new ArrayDeque<>();

        // Insere o simbolo inicial
        pilha.push(simboloInicialPilha);

        int posicao = 0;
        int limite = 10000;
        int passos = 0;

        // Executa as transicoes
        while (passos < limite) {

            passos++;

            // Se toda a palavra foi lida e o estado e final
            if (posicao == palavra.length()
                    && atual.isFinal()) {

                return true;
            }

            if (pilha.isEmpty()) {
                return false;
            }

            String topo = pilha.peek();

            String simboloAtual = EPSILON;

            if (posicao < palavra.length()) {

                simboloAtual = String.valueOf(
                        palavra.charAt(posicao)
                );
            }

            List<TransicaoPilha> aplicaveis =
                    new ArrayList<>();

            // Procura as transicoes possiveis
            for (TransicaoPilha transicao : transicoesPilha) {

                boolean mesmoEstado =
                        transicao.getOrigem().equals(atual);

                boolean mesmoTopo =
                        transicao.getTopoPilha().equals(topo);

                boolean entradaValida =
                        transicao.getSimboloEntrada().equals(EPSILON)
                        || transicao.getSimboloEntrada()
                                .equals(simboloAtual);

                if (mesmoEstado && mesmoTopo && entradaValida) {

                    aplicaveis.add(transicao);
                }
            }

            // Nenhuma transicao encontrada
            if (aplicaveis.isEmpty()) {
                return false;
            }

            // Impede escolhas ambiguas nesta implementacao
            if (aplicaveis.size() > 1) {
                System.out.println(
                    "Erro: AP deterministico possui transicoes ambiguas!"
                );
                return false;
            }

            TransicaoPilha transicao = aplicaveis.get(0);

            // Remove o topo da pilha
            pilha.pop();

            // Insere os novos simbolos na pilha
            String substituir = transicao.getSubstituirPilha();

            if (!substituir.equals(EPSILON)) {

                // O primeiro caractere representa o novo topo
                for (int i = substituir.length() - 1; i >= 0; i--) {

                    pilha.push(
                        String.valueOf(substituir.charAt(i))
                    );
                }
            }

            // Atualiza o estado
            atual = transicao.getDestino();

            // Avanca na palavra se nao for epsilon
            if (!transicao.getSimboloEntrada().equals(EPSILON)) {
                posicao++;
            }
        }

        System.out.println("Limite de processamento atingido!");
        return false;
    }

    // Exibe o automato
    @Override
    public void exibirAutomato() {

        super.exibirAutomato();

        System.out.println(
            "\nSimbolo inicial da pilha: "
            + simboloInicialPilha
        );

        exibirTransicoesPilha();
    }
}
